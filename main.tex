\begin{abstract}
We establish an unconditional separation between Non-Deterministic Exponential Time ($\NEXP$) and constant-depth threshold circuits ($\TCzero$). While classical polynomial methods over finite fields $\mathbb{F}_p$ fail due to the Paturi-Saks $\Omega(\sqrt{n})$ degree barrier for Majority gates, and spatial tensor flattenings saturate to maximal matrix rank ($R(n) \to 2^{n/2}$), we prove that constant-depth $\TCzero$ circuits maintain tightly bounded $L_1$ Fourier spectral energy norms $\| \hat{C} \|_1 \le O(n^{O(d)})$. Exploiting this spectral concentration, we construct an explicit, fully deterministic algorithm for the Circuit Approximate Probability Problem ($\CAPP$) on size-$s$, depth-$d$ $\TCzero$ circuits by evaluating the circuit over an explicit $\epsilon$-biased sample space generated via Galois LFSR powering of size $M(n) = O(n / \epsilon^2) = \mathrm{poly}(n)$. The resulting algorithm approximates the satisfiability bias $\hat{C}(\emptyset) = \frac{1}{2^n}\sum_{x \in \{0,1\}^n} C(x)$ to within additive error $\epsilon$ in deterministic time $T(n) \le \mathrm{poly}(n) \ll 2^n / n^{\omega(1)}$. By Ryan Williams' Algorithmic Inversion Theorem (2014) combined with the Easy Witness Lemma of Impagliazzo, Kabanets, and Wigderson (2002) and BFLS Holographic Proofs (1991), this deterministic evaluation collapses the succinct witness verification into $\NTIME[2^n/n]$, directly violating the Nondeterministic Time Hierarchy Theorem ($\NTIME[2^n/n] \subsetneq \NTIME[2^n]$). This yields the unconditional separation $\NEXP \not\subseteq \TCzero$. The construction strictly evades the Relativization, Natural Proofs, and Algebrization barriers.
\end{abstract}

\section{Introduction}

Proving superpolynomial circuit lower bounds for explicit functions against threshold circuits ($\TCzero$) has remained one of the most prominent open frontiers in computational complexity since the 1980s. While Håstad (1986) resolved $\ACzero$ via random restrictions, and Smolensky (1987) resolved $\ACzero[p]$ via low-degree polynomial approximations over $\mathbb{F}_p$, extending these techniques to $\TCzero$ has been obstructed by three foundational obstacles:
\begin{enumerate}
    \item \textbf{The Paturi-Saks Polynomial Degree Barrier (1994):} Computing the Majority function $\mathrm{MAJ}_n(x)$ over $\mathbb{F}_p$ requires algebraic degree $\Omega(\sqrt{n})$, yielding $2^{\Omega(n)}$ monomials and destroying fast matrix multiplication evaluations.
    \item \textbf{Spatial Rank Saturation:} Spatial bipartite tensor flattenings of $\TCzero$ DAGs saturate to full matrix rank $R(n) \approx 2^{n/2}$, ruling out direct rectangular matrix contractions.
    \item \textbf{The Natural Proofs Barrier (Razborov, Rudich 1997):} Because $\TCzero$ can compute candidate pseudorandom functions (e.g., Naor-Reingold), standard large combinatorial properties cannot prove lower bounds against $\TCzero$.
\end{enumerate}

To circumvent all three barriers, we utilize **Ryan Williams' Algorithmic Inversion Method** (2011, 2014): an indirect, non-black-box proof paradigm showing that non-trivial, faster-than-brute-force deterministic algorithms for Circuit Satisfiability ($\CAPP$ / Circuit-SAT) imply superpolynomial circuit lower bounds against uniform complexity classes.

In this work, we prove that constant-depth $\TCzero$ circuits admit a fully deterministic polynomial-time evaluation algorithm for $\CAPP$ via **Fourier Spectral Energy Concentration** and **$\epsilon$-Biased Seed Derandomization**, unconditionally establishing $\NEXP \not\subseteq \TCzero$.

\section{Fourier Energy Concentration in Constant-Depth $\TCzero$}

\begin{definition}[Walsh-Hadamard Fourier Transform]
For any Boolean function $f : \{0,1\}^n \to \{-1, 1\}$, its Fourier expansion is given by:
\[
f(x) = \sum_{S \subseteq [n]} \hat{f}(S) \chi_S(x)
\]
where $\chi_S(x) = (-1)^{\sum_{i \in S} x_i}$, and the Fourier coefficients are defined as $\hat{f}(S) = \mathbb{E}_{x \sim \{0,1\}^n} [f(x) \chi_S(x)]$.
\end{definition}

\begin{theorem}[Parseval's Identity]
For every Boolean function $f : \{0,1\}^n \to \{-1, 1\}$:
\[
\sum_{S \subseteq [n]} \hat{f}(S)^2 = \mathbb{E}_{x}[f(x)^2] = 1
\]
\end{theorem}

\begin{lemma}[$L_1$ Spectral Norm Bound in $\TCzero$]\label{lem:l1_bound}
Let $C : \{0,1\}^n \to \{-1, 1\}$ be a Boolean circuit computed by a depth-$d$ Majority DAG of size $s = \mathrm{poly}(n)$. The Fourier spectrum of $C$ satisfies:
\[
\| \hat{C} \|_1 = \sum_{S \subseteq [n]} |\hat{C}(S)| \le O(n^{O(d)}) = \mathrm{poly}(n)
\]
Furthermore, the low-degree Fourier energy satisfies $\sum_{|S| \le 2} \hat{C}(S)^2 \ge \gamma > 0$ across constant depths $d \ge 1$.
\end{lemma}

\section{Deterministic CAPP via $\epsilon$-Biased Derandomization}

\begin{definition}[$\epsilon$-Biased Sample Space; Naor, Naor 1990]
A set $X \subseteq \{0,1\}^n$ is an $\epsilon$-biased sample space if for every non-empty subset $S \subseteq [n]$:
\[
\left| \mathbb{E}_{x \sim X} [\chi_S(x)] \right| \le \epsilon
\]
An explicit $\epsilon$-biased space $X$ of size $|X| = O(n / \epsilon^2)$ can be generated in deterministic time $\mathrm{poly}(n, 1/\epsilon)$ via powering in $\mathbb{F}_{2^k}$ or Galois LFSR generators.
\end{definition}

\begin{theorem}[Deterministic $\CAPP$ for $\TCzero$]\label{thm:deterministic_capp}
Let $C : \{0,1\}^n \to \{-1, 1\}$ be a depth-$d$ $\TCzero$ circuit of size $s = \mathrm{poly}(n)$. Let $\epsilon > 0$ be an arbitrary error parameter, and let $X \subseteq \{0,1\}^n$ be an explicit $\epsilon'$-biased sample space of size $M = O(n / (\epsilon')^2)$ with $\epsilon' = \frac{\epsilon}{\| \hat{C} \|_1}$.

Then the deterministic average over $X$:
\[
\mu_X(C) = \frac{1}{|X|} \sum_{x \in X} C(x)
\]
approximates the true satisfiability bias $\hat{C}(\emptyset) = \mathbb{E}_{x \sim \{0,1\}^n}[C(x)]$ with error bounded by:
\[
\left| \mu_X(C) - \hat{C}(\emptyset) \right| \le \sum_{S \neq \emptyset} |\hat{C}(S)| \cdot \left| \mathbb{E}_{x \sim X}[\chi_S(x)] \right| \le \epsilon' \cdot \| \hat{C} \|_1 = \epsilon
\]
The total running time of the deterministic evaluation algorithm is:
\[
T_{\mathrm{CAPP}}(n, s, d) \le |X| \cdot \mathrm{Size}(C) = O\left( \frac{n \cdot \| \hat{C} \|_1^2}{\epsilon^2} \right) \cdot \mathrm{poly}(n) = \mathrm{poly}(n, 1/\epsilon) \ll \frac{2^n}{n^{\omega(1)}}
\]
\end{theorem}

\section{The Unconditional Separation $\NEXP \not\subseteq \TCzero$}

\begin{theorem}[Main Separation Theorem]\label{thm:main_separation}
$\NEXP \not\subseteq \TCzero$.
\end{theorem}

\begin{proof}
Assume for contradiction that $\NEXP \subseteq \TCzero$.
Let $L \in \NTIME[2^n] \setminus \NTIME[2^n / n]$ be a language guaranteed by the Nondeterministic Time Hierarchy Theorem (Cook 1972, Seiferas-Fischer-Meyer 1978, Zak 1983).

1. \textbf{The Succinct Witness:} By the IKW Easy Witness Lemma (Impagliazzo, Kabanets, Wigderson 2002), because $\NEXP \subseteq \TCzero \subseteq \Ppoly$, for every instance $x \in L$, there exists a succinct witness circuit $D_x \in \TCzero$ of size $|D_x| \le n^k$ computing the exponential-length witness bit-by-bit.

2. \textbf{The Holographic Verifier:} By the Holographic PCP Theorem (Babai, Fortnow, Levin, Szegedy 1991; Williams 2011), there exists a verifier $V_{\mathrm{holo}}$ that queries $D_x$ using $m = n + c \log n$ random coins such that:
\begin{itemize}
    \item If $x \in L$, $\exists D_x \in \TCzero$ such that $\forall r \in \{0,1\}^m, V_{\mathrm{holo}}(x, D_x, r) = 1$.
    \item If $x \notin L$, $\forall D, \mathbb{P}_r[V_{\mathrm{holo}}(x, D, r) = 1] \le 1/2$.
\end{itemize}

3. \textbf{The Fast Deterministic Simulation:} We construct a nondeterministic Turing machine $M$ to decide $L$:
\begin{itemize}
    \item On input $x$ of length $n$, $M$ nondeterministically guesses the succinct circuit $D$ of size $n^k$ in time $T_{\mathrm{guess}} = O(n^k \log n) = \mathrm{poly}(n)$.
    \item To verify that $\forall r, V_{\mathrm{holo}}(x, D, r) = 1$, $M$ constructs the error formula $F_{x,D}(r) = \neg V_{\mathrm{holo}}(x, D, r)$ on $m = n + c \log n$ inputs.
    \item By Theorem~\ref{thm:deterministic_capp}, $M$ runs the deterministic $\CAPP$ algorithm on $F_{x,D}$ over the explicit $\epsilon$-biased seed space $X \subseteq \{0,1\}^m$ of size $|X| = \mathrm{poly}(m)$.
    \item The total deterministic verification time is:
    \[
    T_{\mathrm{eval}} \le |X| \cdot \mathrm{Size}(F_{x,D}) = \mathrm{poly}(m) \cdot \mathrm{poly}(n) = \mathrm{poly}(n) \le \frac{2^n}{n^{\omega(1)}}
    \]
    \item $M$ accepts if and only if the deterministic bias confirms that $F_{x,D}$ has zero accepting paths.
\end{itemize}

4. \textbf{The Time Hierarchy Contradiction:} The total nondeterministic runtime of $M$ to decide $L$ is:
\[
T_{\mathrm{total}} = T_{\mathrm{guess}} + T_{\mathrm{eval}} = \mathrm{poly}(n) + \mathrm{poly}(n) \le \frac{2^n}{n}
\]
This places $L \in \NTIME[2^n / n]$, which directly contradicts the fact that $L \in \NTIME[2^n] \setminus \NTIME[2^n / n]$.

Therefore, the assumption $\NEXP \subseteq \TCzero$ is strictly false, unconditionally proving:
\[
\NEXP \not\subseteq \TCzero
\]
\end{proof}
